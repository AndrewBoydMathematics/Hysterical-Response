\begin{abstract}
The galactic-modelling companion derives the baryonic Tully--Fisher (BTFR) exponent
$\vf^{4}\propto\Mb$ from a capacity attractor, but leaves the absolute acceleration
scale $a_0$ encoded in a combination of response couplings.
This paper closes that zero point.
Hysterical Response is causal memory of an already-existing interaction; the softest
gravitational residual in an expanding universe is IR-pinned by cosmological-horizon
KMS memory with Euclidean period $\beta=2\pi/H_0$.
The associated memory acceleration is $a_{\mathrm{IR}}=c/\beta=H_0c/2\pi$.
Matching the mature BTFR writing $\vf^{4}=Ga_0\Mb$ to that scale yields
\begin{equation*}
  \boxed{a_0=\frac{H_0c}{2\pi}.}
\end{equation*}
The factor $2\pi$ is horizon regularity, not a MOND fitting constant.
The scale is a property of the residual gravitational channel and the FLRW congruence,
not of any particular probe's age.
No new force carrier is introduced: the result normalizes ordinary gravity's Hysterical
leftover, organized as the galactic halo of the companion paper.
\end{abstract}

\paragraph{Ontology.}
The response-theory companion defines the stationary Hysterical response~\cite{boyd2026response}
\begin{equation}
\label{eq:FH}
\FH=-\Tr\!\bigl(F\,\mathcal L^{-1}\mathcal S\bigr)
=\int_0^\infty\Tr\!\bigl(F\,e^{t\mathcal L}\mathcal S\bigr)\,\dd t

\section{Inherited BTFR structure}
\label{sec:btfr}
The galactic-modelling companion supplies the ballistic residual halo
$g_{\mathrm{resp}}=CQ/r$ and, under capacity--filling dynamics, the mature attractor~\cite{boyd2026galactic}
\begin{equation}
\label{eq:K}
\vf^{4}=K\Mb,\qquad K=\frac{C^{2}A^{2}}{y_{\ast}}.
\end{equation}
Writing the empirical BTFR as $\vf^{4}=Ga_0\Mb$~\cite{mcgaugh2012,lelli2016} identifies
\begin{equation}
\label{eq:Ga0}
Ga_0=K
\end{equation}
on that attractor.
The \emph{exponent} $\vf^{4}\propto\Mb$ is therefore already theorem material there;
the present paper fixes only the absolute scale $a_0=K/G$.

\begin{assumption}[Mature attractor normalization]
\label{ass:attractor}
We work on the mature BTFR attractor~\eqref{eq:K}--\eqref{eq:Ga0} of~\cite{boyd2026galactic},
so a single positive acceleration $a_0$ normalizes all mature response-dominated circular speeds
through $\vf^{4}=Ga_0\Mb$.
\end{assumption}

\section{Horizon memory period}
\label{sec:horizon}
Hysterical Response is history-dependent through the resolvent integral~\eqref{eq:FH}.
At a causal horizon the same structure is packaged as retarded membrane memory~\cite{boyd2026horizon}.

\begin{literaturehypothesis}[Euclidean horizon period]
\label{lit:period}
For a nonextremal Killing horizon with surface gravity $\kappa>0$, Euclidean regularity
requires $\tau\sim\tau+2\pi/\kappa$, hence a KMS period~\cite{Hawking1975,Unruh1976}
\begin{equation}
\label{eq:beta}
\beta=\frac{2\pi}{\kappa}
\end{equation}
(geometric units $\hbar=c=k_B=1$; $c$ is restored below).
\end{literaturehypothesis}

\begin{literaturehypothesis}[Cosmological horizon surface gravity]
\label{lit:cosmo-kappa}
In de~Sitter (or late quasi--de~Sitter) geometry the cosmological event horizon has
surface gravity $\kappa=H$~\cite{GibbonsHawking1977}.
At the present epoch we set $\kappa=H_0$.
\end{literaturehypothesis}

\begin{remark}[Same $2\pi$ as black-hole HRT]
The horizon-response companion uses precisely~\eqref{eq:beta} to obtain
$T_H=\kappa/(2\pi)$ and the Hawking occupation factor~\cite{boyd2026horizon}.
The cosmological application here uses the same geometric period as the memory clock
of the residual gravitational channel, not a new fitting constant.
\end{remark}

\section{IR memory acceleration}
\label{sec:aIR}
\begin{definition}[IR memory acceleration]
\label{def:aIR}
Restoring $c$, the acceleration associated with the horizon memory period~\eqref{eq:beta} is
\begin{equation}
\label{eq:aIR}
a_{\mathrm{IR}}:=\frac{c}{\beta}=\frac{c\kappa}{2\pi}.
\end{equation}
\end{definition}

\begin{theorem}[Cosmological memory scale]
\label{thm:aIR}
Under Literature Hypotheses~\ref{lit:period}--\ref{lit:cosmo-kappa} and Definition~\ref{def:aIR}
with $\kappa=H_0$,
\begin{equation}
\label{eq:aIR-H0}
\boxed{a_{\mathrm{IR}}=\frac{H_0c}{2\pi}.}
\end{equation}
\end{theorem}

\begin{proof}
Substitute $\beta=2\pi/H_0$ into~\eqref{eq:aIR}.
\end{proof}

\begin{remark}[Why not $cH_0$?]
Raw Hubble time $t_H=1/H_0$ would give $c/t_H=cH_0$.
Horizon regularity replaces $t_H$ by the Euclidean/KMS period $\beta=2\pi/H_0$, which is
smaller by $2\pi$ and matches the empirical BTFR scale~\cite{mcgaugh2012,lelli2016}
at $H_0\sim70\,\mathrm{km\,s^{-1}\,Mpc^{-1}}$.
Equating Unruh temperature $a/(2\pi)$ to Gibbons--Hawking $H/(2\pi)$ instead yields
$a=cH_0$, which is the wrong empirical factor; the HRT reading uses memory time $\beta$,
not that Unruh identification.
\end{remark}

\section{Subject independence}
\label{sec:subject}
\begin{proposition}[Subject independence of the IR floor]
\label{prop:subject}
The scale~\eqref{eq:aIR-H0} depends only on the FLRW congruence (through $H_0$) and on $c$.
It does not depend on the proper age of a particular probe, nor on a passive Hilbert-space
relabelling of that probe.
\end{proposition}

\begin{remark}
A newly formed grain falls in a preexisting residual potential $\Phi_H$; it does not reset
the channel memory that built $\Phi_H$.
Basis covariance of Hysterical Response already forbids passive subject-relabelling from
changing $\FH$~\cite{boyd2026response}.
Homogeneity makes $H_0$ shared by every comoving observer at the present epoch.
Local environments (Solar System, dense discs) may have faster gaps and different criticality;
they do not invent a second universal cosmological floor~\cite{boyd2026late}.
\end{remark}

\section{Far-halo IR pin}
\label{sec:pin}
The galactic ballistic halo supports a flat $g_{\mathrm{resp}}\propto1/r$ regime only inside
the propagation length $\lambda=v/\Gamma$~\cite{boyd2026galactic}.
Extended flat curves therefore require a very small far-halo loss rate $\Gamma$.

\begin{assumption}[Far-halo IR pin]
\label{ass:halo-IR}
The softest environmentally allowed gap for the residual gravitational channel in an
expanding universe is the cosmological-horizon memory rate of Literature
Hypotheses~\ref{lit:period}--\ref{lit:cosmo-kappa}.
The far-halo residual is IR-pinned to that horizon mode, so the characteristic acceleration
of the mature residual equals $a_{\mathrm{IR}}$.
\end{assumption}

\begin{remark}
Assumption~\ref{ass:halo-IR} is the physical bridge from horizon geometry to galactic
normalization.
It is not a corollary of ballistic transport alone; it is the statement that the soft pole
organizing the mature halo is the cosmologically cut-off gravitational residual, not a
private galactic clock.
\end{remark}

\section{Zero-point matching and main theorem}
\label{sec:main}
\begin{assumption}[Zero-point matching]
\label{ass:zero-match}
On the mature attractor of Assumption~\ref{ass:attractor}, the BTFR acceleration equals the
IR memory scale:
\begin{equation}
\label{eq:match}
a_0=a_{\mathrm{IR}}.
\end{equation}
Equivalently $K=Ga_{\mathrm{IR}}$, fixing the combination $C^{2}A^{2}/y_{\ast}$ without resolving
$C$ and $A$ separately.
\end{assumption}

\begin{theorem}[BTFR zero point from horizon memory]
\label{thm:a0}
Assume Assumptions~\ref{ass:attractor},~\ref{ass:halo-IR}, and~\ref{ass:zero-match}, together
with Theorem~\ref{thm:aIR}.
Then
\begin{equation}
\label{eq:a0}
\boxed{a_0=\frac{H_0c}{2\pi},}
\qquad
\vf^{4}=G\frac{H_0c}{2\pi}\,\Mb
\qquad\text{on the mature attractor.}
\end{equation}
\end{theorem}

\begin{proof}
Assumption~\ref{ass:zero-match} gives $a_0=a_{\mathrm{IR}}$; Theorem~\ref{thm:aIR} gives
$a_{\mathrm{IR}}=H_0c/2\pi$; Assumption~\ref{ass:attractor} with $K=Ga_0$ gives the velocity form.
\end{proof}

\begin{corollary}[Numerical consistency]
\label{cor:numeric}
For $H_0\simeq67$--$70\,\mathrm{km\,s^{-1}\,Mpc^{-1}}$~\cite{Planck2018},
equation~\eqref{eq:a0} yields $a_0\simeq(1.05$--$1.12)\times10^{-10}\,\mathrm{m\,s^{-2}}$,
compatible with the gas-rich BTFR calibration
$a_0\simeq1.20\times10^{-10}\,\mathrm{m\,s^{-2}}$~\cite{mcgaugh2012,lelli2016} at the
precision of the present matching hypotheses.
\end{corollary}

\section{Relation to MOND phenomenology}
\label{sec:mond}
MOND/RAR phenomenology uses a universal acceleration $a_0$ inside an interpolating map
$g(g_b;a_0)$~\cite{McGaugh2016}.
Hysterical Response uses a different ontology: residual ordinary gravity with ballistic
halo $g_{\mathrm{resp}}=CQ/r$ and capacity attractor~\cite{boyd2026galactic}.
Theorem~\ref{thm:a0} explains why an effective $a_0$ of cosmological size appears, without
adopting the MOND force law.
Finite-radius mappings therefore need not coincide; only the outer zero point is fixed here.

\section{Conclusion}
\label{sec:conclusion}
The galactic BTFR exponent is settled by capacity dynamics; the absolute zero point is
settled by cosmological-horizon Hysterical memory.
With Euclidean period $\beta=2\pi/H_0$ and far-halo IR pinning of the residual gravitational
channel,
\[
\boxed{a_0=\frac{H_0c}{2\pi}.}
\]
The $2\pi$ is horizon regularity; the subject of the memory is the residual channel on the
FLRW congruence, not a particular probe.
What remains for phenomenology is coexistence checks against galactic fits and Solar null
tests that use this normalization, not a second independent universal acceleration.
