\begin{abstract}
Hysterical Response Theory (HRT) isolates the residual response of an already-existing
interaction under open-system dynamics,
$\FH=-\Tr(F\mathcal L^{-1}\mathcal S)$~\cite{boyd2026response}.
Applied at a nonextremal stationary horizon through the membrane / open-system
embedding of the exterior~\cite{Thorne1986,Parikh1999,HuRavalSinha2002}, that package
reproduces analytically the standard semiclassical predictions of Hawking
radiation~\cite{Hawking1975} and Tolman local temperature~\cite{Tolman1930,TolmanEhrenfest1930}.
The Euclidean period fixed by horizon regularity supplies
$T_H=\kappa/(2\pi)$; quantum fluctuation--dissipation and detailed balance yield the
Planck occupation; exterior greybodies filter the asymptotic flux; static equilibrium
continues the same KMS scale to
$T_{\mathrm{loc}}(r)=T_H/\sqrt{1-2M/r}$ for $r>2M$.
We record the structural dictionary that makes this an effective-method equivalence, not
merely a similarity of shape, and interpret why Hawking (fluctuation branch at a causal
dissipative boundary) and Tolman (static equilibrium profile of that thermal response)
each touched the Hysterical Response without naming it.
Geometry-free derivation of the Euclidean period from a Hysterical action alone is not
claimed; Hawking likewise takes that period from horizon regularity.
\end{abstract}

\section{Scope and status}
\label{sec:scope}
The response-theory paper defines the Hysterical Response as the residual component of
an already-existing interaction under exponentially stable reduced
dynamics~\cite{boyd2026response}:
\begin{equation}
\label{eq:FH}
\boxed{\FH:=-\Tr\!\bigl(F\,\mathcal L^{-1}\mathcal S\bigr).}
\end{equation}
No new force carrier is introduced.
The present paper asks what that package predicts when the interaction channel is a
quantum field on a Schwarzschild exterior with the horizon represented as a causal,
dissipative boundary.

The claim is
\begin{equation}
\label{eq:claim}
\boxed{\begin{gathered}
\text{HRT horizon response}
\;+\;
\text{membrane/open-system embedding}
\;+\;
\text{horizon regularity}\\
\Longrightarrow\quad
\text{Hawking asymptotics and Tolman static temperature}
\quad\text{(analytic agreement).}
\end{gathered}}
\end{equation}

\section{Horizon regularity fixes the response temperature}
\label{sec:geometry}
\begin{assumption}[Nonextremal stationary horizon]
\label{ass:horizon}
The spacetime admits a nonextremal Killing horizon with surface gravity $\kappa>0$.
Near the horizon the metric reduces locally to Rindler form
\begin{equation}
\label{eq:rindler}
\dd s^2\simeq-(\kappa\rho)^2\,\dd t^2+\dd\rho^2+r_H^2\,\dd\Omega^2.
\end{equation}
\end{assumption}

\begin{literaturehypothesis}[Euclidean regularity]
\label{lit:period}
Under $t=-i\tau$, regularity at $\rho=0$ requires the identification
$\tau\sim\tau+2\pi/\kappa$~\cite{Hawking1975,Unruh1976}.
A regular horizon state is therefore KMS at inverse temperature
$\beta_H=2\pi/\kappa$ (units $\hbar=c=k_B=1$).
\end{literaturehypothesis}

\begin{theorem}[Hawking temperature from the Euclidean period]
\label{thm:temp}
If $\beta_H=2\pi/\kappa$ with $\kappa\neq0$, then
\begin{equation}
\label{eq:temp}
T_H:=\beta_H^{-1}=\frac{\kappa}{2\pi}.
\end{equation}
For Schwarzschild, $\kappa=1/(4M)$ yields $T_H=1/(8\pi M)$.
\end{theorem}

\begin{proof}
Immediate from $\beta_H=2\pi/\kappa$.
\end{proof}

\begin{remark}[Shared geometric input]
\label{rem:geom}
Equation~\eqref{eq:temp} uses horizon geometry exactly as in Hawking's derivation.
We do not claim to derive $\beta_H$ from a Hysterical action with the metric deleted.
\end{remark}

\section{Causal horizon memory as Hysterical response}
\label{sec:memory}
\begin{literaturehypothesis}[Membrane / open-system embedding]
\label{lit:membrane}
The exterior influence of the horizon is represented by a stretched, dissipative
membrane~\cite{Thorne1986,Parikh1999}.
Influence-functional treatments of black-hole backreaction likewise yield nonlocal
dissipation and stochastic noise together, with the noise linked to Bogoliubov
particle production~\cite{HuRavalSinha2002}.
\end{literaturehypothesis}

\begin{model}[Horizon Langevin response]
\label{mod:langevin}
For a response coordinate $h$ (e.g.\ an electromagnetic multipole on the stretched
horizon),
\begin{equation}
\label{eq:langevin}
\dot h(t)+\int_{-\infty}^{t}\Gamma(t-t')h(t')\,\dd t'
=\sigma A_{\mathrm{in}}(t)+\xi(t),
\end{equation}
with causal memory kernel $\Gamma$, incoming drive $A_{\mathrm{in}}$, and intrinsic
noise $\xi$.
In frequency space,
\begin{equation}
\label{eq:chi}
h(\omega)=\chi_R(\omega)\bigl[\sigma A_{\mathrm{in}}(\omega)+\xi(\omega)\bigr],
\qquad
A_{\mathrm{out}}(\omega)=\alpha(\omega)h(\omega).
\end{equation}
\end{model}

\begin{proposition}[Hysterical identification]
\label{prop:dict}
Equation~\eqref{eq:langevin} is Hysterical in the sense of~\cite{boyd2026response}:
the present response depends on the previous trajectory through a retarded kernel.
The susceptibility $\chi_R$ is the open-system linear response of the horizon channel;
$\xi$ is the fluctuation partner of dissipation, not an externally inserted Hawking
flux.
\end{proposition}

\section{Detailed balance produces the Hawking factor}
\label{sec:hawking}
\begin{literaturehypothesis}[Quantum FDR at the horizon period]
\label{lit:fdr}
With $\beta_H$ from Literature Hypothesis~\ref{lit:period}, the unsymmetrized noise
obeys the KMS detailed-balance relation
\begin{equation}
\label{eq:kms}
S_\xi(-\omega)=e^{-\beta_H\omega}\,S_\xi(+\omega),
\end{equation}
equivalently a quantum fluctuation--dissipation relation tying the symmetrized noise to
$\Imop\Gamma_R$~\cite{HuRavalSinha2002,Unruh1976}.
\end{literaturehypothesis}

\begin{lemma}[Detailed-balance occupation]
\label{lem:occupation}
If $r\neq1$, $p_+\neq0$, and $p_-=r\,p_+$, then
\[
\frac{p_-}{p_+-p_-}=\frac{r}{1-r}.
\]
\end{lemma}

\begin{proof}
Substitute $p_-=r\,p_+$ and clear the common factor.
\end{proof}

\begin{lemma}[Exponential ratio is Bose]
\label{lem:bose}
If $e^{x}\neq1$, then
\[
\frac{e^{-x}}{1-e^{-x}}=\frac{1}{e^{x}-1}.
\]
\end{lemma}

\begin{proof}
Rewrite $e^{-x}=1/e^{x}$ and clear denominators.
\end{proof}

\begin{theorem}[Hawking occupation from HRT detailed balance]
\label{thm:hawking-occupation}
Under Literature Hypotheses~\ref{lit:period}--\ref{lit:fdr} and
Model~\ref{mod:langevin}, set $r=e^{-\beta_H\omega}$.
Then the detailed-balance occupation is
\begin{equation}
\label{eq:bose}
n_\omega=\frac{1}{e^{\beta_H\omega}-1}.
\end{equation}
The asymptotic number flux takes the standard form
\begin{equation}
\label{eq:flux}
\frac{\dd N}{\dd t\,\dd\omega}
=\frac{1}{2\pi}\sum_{\ell,m,p}
\frac{\mathcal G_{\ell mp}(\omega)}{e^{\beta_H\omega}-1},
\end{equation}
where $\mathcal G$ is the exterior greybody transmission factor.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:occupation} with $r=e^{-\beta_H\omega}$ and Lemma~\ref{lem:bose} with
$x=\beta_H\omega$ give~\eqref{eq:bose}.
Greybodies encode propagation through the exterior curvature potential and multiply
the universal thermal denominator fixed by the horizon response~\cite{Hawking1975}.
\end{proof}

\begin{theorem}[Effective-method agreement with Hawking]
\label{thm:hawking-equiv}
Under the membrane/open-system embedding (Literature Hypothesis~\ref{lit:membrane}) and
the horizon KMS/FDR package (Literature Hypotheses~\ref{lit:period},~\ref{lit:fdr}),
the HRT horizon-response calculation of Theorem~\ref{thm:hawking-occupation} agrees
analytically with Hawking's asymptotic thermal prediction:
same $T_H=\kappa/(2\pi)$, same Planck factor, and same greybody-filtered flux.
At this effective-response abstraction the two methods are the same calculation, not
merely similar shapes.
\end{theorem}

\begin{proof}
Hawking's Euclidean/Bogoliubov analysis supplies precisely the period $\beta_H$, the
thermal detailed-balance ratio, and the exterior transmission problem that enter
Theorem~\ref{thm:hawking-occupation}~\cite{Hawking1975,HuRavalSinha2002}.
The HRT packaging identifies those objects with the noise and susceptibility of
Definition~\ref{def:dict}.
The algebraic output is identical.
\end{proof}

\begin{remark}[Mean versus covariance]
\label{rem:mean}
At fixed mass the classical quasinormal poles remain stable.
Hawking production is growth of quantum covariance,
$\langle A_{\mathrm{out}}\rangle=0$ while
$\langle A_{\mathrm{out}}^{\dagger}A_{\mathrm{out}}\rangle=n_\omega>0$,
not exponential growth of a classical mean field.
\end{remark}

\section{Tolman exterior temperature}
\label{sec:tolman}
\begin{literaturehypothesis}[Tolman equilibrium law]
\label{lit:tolman}
In a static gravitational field, thermal equilibrium requires
$T_{\mathrm{loc}}\sqrt{-g_{tt}}=\mathrm{const}$~\cite{Tolman1930,TolmanEhrenfest1930}.
\end{literaturehypothesis}

\begin{assumption}[Static exterior observers]
\label{ass:static}
Restrict to Schwarzschild $r>2M$ and static observers, for whom
$-g_{tt}=1-2M/r>0$.
\end{assumption}

\begin{theorem}[Tolman local temperature from the horizon scale]
\label{thm:tolman}
Under Literature Hypothesis~\ref{lit:tolman}, Assumption~\ref{ass:static}, and
Theorem~\ref{thm:temp} with asymptotic normalization $T_\infty=T_H$,
\begin{equation}
\label{eq:tolman}
T_{\mathrm{loc}}(r)=\frac{T_H}{\sqrt{1-2M/r}}\qquad(r>2M).
\end{equation}
\end{theorem}

\begin{proof}
Tolman's law gives $T_{\mathrm{loc}}(r)\sqrt{1-2M/r}=T_\infty$.
The Hartle--Hawking / horizon-KMS state that underlies
Theorem~\ref{thm:hawking-equiv} has $T_\infty=T_H$.
\end{proof}

\begin{theorem}[Effective-method agreement with Tolman]
\label{thm:tolman-equiv}
Once the HRT/Hawking package fixes the single static-Killing temperature scale $T_H$
at infinity, Tolman's equilibrium law continues that scale to every static radius
$r>2M$ by~\eqref{eq:tolman}.
The HRT horizon response therefore agrees analytically with Tolman's local-temperature
prediction for static observers.
\end{theorem}

\begin{proof}
Immediate from Theorems~\ref{thm:hawking-equiv} and~\ref{thm:tolman}.
\end{proof}

\begin{remark}[Unruh acceleration versus Tolman]
\label{rem:unruh}
The proper acceleration of a static Schwarzschild observer is
$a(r)=(M/r^2)/\sqrt{1-2M/r}$.
The naive Unruh assignment $a/(2\pi)$ equals~\eqref{eq:tolman} only in the near-horizon
Rindler limit.
Away from the horizon the equilibrium temperature of the static KMS state is the Tolman
profile, not $a/(2\pi)$ at every $r$~\cite{Unruh1976,Tolman1930}.
\end{remark}

\begin{proposition}[No static Tolman temperature for $r<2M$]
\label{prop:interior}
For $r<2M$ the Killing field $\partial_t$ is spacelike, so there are no static observers
and~\eqref{eq:tolman} does not apply.
Interior radiative behaviour (partner modes, free-fall regularity) is outside the
static Tolman claim; free-fall regularity near the horizon is the standard semiclassical
expectation~\cite{Unruh1976}.
\end{proposition}

\section{Why Hawking and Tolman touched the Hysterical Response}
\label{sec:interpret}
This section is interpretive: it explains the contact, and is not Lean-certified.

\paragraph{Hawking.}
Hawking analyzed quantum fields against a causal horizon that absorbs and transmits,
with vacuum regularity fixing a thermal scale~\cite{Hawking1975}.
That arena---a dissipative boundary, a retarded response, and a fluctuation partner
required by detailed balance---is exactly the open-system response setting in which
HRT is defined~\cite{boyd2026response,HuRavalSinha2002}.
He computed the asymptotic occupation of the fluctuation branch of that boundary
response.
HRT names the same structure as horizon Hysterical memory plus noise
(Definition~\ref{def:dict}); the Euclidean period is the shared geometric input
(Remark~\ref{rem:geom}).

\paragraph{Tolman.}
Tolman asked for thermal equilibrium in a static gravitational
field~\cite{Tolman1930,TolmanEhrenfest1930}.
Given any theory that supplies a single static-Killing temperature $T_H$ at infinity,
his law is the unique static radial continuation.
He therefore touched the spatial organization of the same equilibrium response whose
fluctuation scale Hawking (and the HRT horizon package) fix at the boundary---without
needing the membrane Langevin vocabulary.

\paragraph{Joint reading.}
Hawking fixed the fluctuation scale at the causal dissipative boundary; Tolman fixed
the static radial profile of that equilibrium.
HRT's horizon package is the open-system rephrasing that joins both, which is why their
results agree analytically with HRT predictions
(Theorems~\ref{thm:hawking-equiv},~\ref{thm:tolman-equiv}).

\section{Backreaction as feedback}
\label{sec:backreact}
\begin{model}[Semiclassical mass loss]
\label{mod:mass}
To leading scaling order, $P_H\propto A_H T_H^4$ with $A_H\propto M^2$ and
$T_H\propto M^{-1}$, so
\begin{equation}
\label{eq:massloss}
\dot M=-\frac{C}{M^2},\qquad C>0.
\end{equation}
\end{model}

\begin{theorem}[Integrated evaporation law]
\label{thm:evap}
The solution of~\eqref{eq:massloss} with $M(0)=M_0$ satisfies
$M(t)^3=M_0^3-3Ct$ on the interval where $M>0$, and reaches $M=0$ at
$t=M_0^3/(3C)$ when $C\neq0$.
\end{theorem}

\begin{proof}
Separate variables: $M^2\,\dd M=-C\,\dd t$, integrate, and evaluate the extinction time.
\end{proof}

\begin{proposition}[Hotter and more luminous at smaller mass]
\label{prop:feedback}
If $\theta,C,M_{\mathrm{small}},M_{\mathrm{large}}>0$ and $M_{\mathrm{small}}<M_{\mathrm{large}}$,
then $\theta/M_{\mathrm{large}}<\theta/M_{\mathrm{small}}$.
If also $M_{\mathrm{small}}^2<M_{\mathrm{large}}^2$, then
$C/M_{\mathrm{large}}^2<C/M_{\mathrm{small}}^2$.
\end{proposition}

\begin{proof}
Clear positive denominators.
\end{proof}

Thus mass loss raises $T_H$ and the radiated power, producing the standard positive-feedback
evaporation runaway.
The semiclassical model does not determine the ultimate endpoint.

\section{Conclusion}
\label{sec:conclusion}
Under the membrane / open-system embedding of a nonextremal horizon, Hysterical Response
Theory agrees analytically with Hawking radiation and Tolman local temperature:
the same $T_H$, the same Planck factor and greybodies, and the same static profile
$T_{\mathrm{loc}}(r)=T_H/\sqrt{1-2M/r}$ for $r>2M$.
At the effective-response abstraction that agreement is method equivalence under shared
geometric and thermal inputs, not a verbal analogy.

Hawking computed the fluctuation branch of a causal dissipative boundary response;
Tolman computed the static equilibrium continuation of that thermal scale.
HRT is the open-system packaging that joins both.
The Euclidean period remains a geometric input---as it is for Hawking---so
geometry-free thermality from a Hysterical action alone is left open and is not required
for the agreement proved here.
